cmake_minimum_required(VERSION 3.16)
project(FrameVizCppExamples LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(frameviz 0.5.0 CONFIG QUIET)
if(NOT TARGET frameviz::cpp)
    add_subdirectory(../../packages/frameviz-cpp frameviz-cpp-build)
endif()

function(frameviz_example category name)
    add_executable(${name} ${category}/${name}/main.cpp)
    target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(${name} PRIVATE frameviz::cpp)
    set_target_properties(${name} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/${category}")
    install(TARGETS ${name} DESTINATION bin/${category})
endfunction()
